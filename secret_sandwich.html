PROCEDURE - vivado
=========

1.  Open Vivado 2018.
2.  Click Create New Project.
3.  Enter project name and select location.
4.  Choose RTL Project.
5.  Select device:
        o Family: Artix-7
        o Package: CPG256
        o Speed grade: -1
6.  Add a VHDL source file.
7.  Write the RTL code for Full Adder and Full Subtractor.
8.  Click Check Syntax.
9.  Open RTL Analysis -> Open Elaborated Design.
10. View the schematic.
11. Run Behavioral Simulation.
12. Apply inputs using Force Constant.
13. Observe output waveform.
14. Stop and close simulation.

1. RTL Code for 4-bit Synchronous Counter
==========================================

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_ARITH.ALL;
use IEEE.STD_LOGIC_UNSIGNED.ALL;
entity SynchronousCounter is
    Port (
        clk : in STD_LOGIC;
        reset : in STD_LOGIC;
        q : out STD_LOGIC_VECTOR(3 downto 0)
    );
end SynchronousCounter;
architecture Behavioral of SynchronousCounter is
    signal count : STD_LOGIC_VECTOR(3 downto 0) := "0000";
begin
    process (clk, reset)
    begin
        if reset = '1' then
            count <= "0000";
        elsif rising_edge(clk) then
            count <= count + 1;
        end if;
    end process;
    q <= count;
end Behavioral;

3. Chip-Level Timing Constraints (SDC - Synopsys Design Constraints)
=====================================================================

create_clock -name clk -period 1 -waveform {0 0.5} [get_ports "clk"]
set_clock_transition -rise 0.1 [get_clocks "clk"]
set_clock_transition -fall 0.1 [get_clocks "clk"]
set_clock_uncertainty 0.01 [get_ports "clk"]
set_input_delay -max 0.4 [get_ports "rst"] -clock [get_clocks "clk"]
set_input_delay -max 0.4 [get_ports "m"] -clock [get_clocks "clk"]
set_output_delay -max 0.4 [get_ports "count"] -clock [get_clocks "clk"]
