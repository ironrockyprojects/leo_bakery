import socket

server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server_socket.bind(("0.0.0.0", 8082))   # Bind to all available interfaces on port 8082
server_socket.listen()

print("Waiting for connection...")

connection_socket, address = server_socket.accept()

print(f"Client connected from {address} - Echo Server is active!")

while True:
    data = connection_socket.recv(1024)

    if not data:
        print("Client disconnected.")
        break

    message = data.decode()
    print(f"Received from client: {message}")

    connection_socket.send(data)

    print(f"Echoed back to client: {message}")

connection_socket.close()

<-----echo command client code--->

import socket

client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

client_socket.connect(("127.0.0.1", 8082))   # Connect to server on localhost and port 8082

print("Connected to Echo Server!")

while True:
    message = input("Enter a message (type 'exit' to quit): ")

    if message.lower() == 'exit':
        print("Exiting...")
        break

    # Send the message to the server
    client_socket.send(message.encode())

    # Receive the echoed message from the server
    data = client_socket.recv(1024)

    print(f"Echo from server: {data.decode()}")

client_socket.close()


<-----ping command server code---->

import socket

def start_server(host='127.0.0.1', port=12345):
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
        s.bind((host, port))
        print(f"UDP Server running on {host}:{port}")

        while True:
            data, addr = s.recvfrom(1024)
            print(f"Received message from {addr}: {data.decode()}")
            s.sendto(b'Ping', addr)

if __name__ == "__main__":
    start_server()

<------ping command client code---->

import socket
import time

def ping_server(host='127.0.0.1', port=12345):
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
        try:
            s.settimeout(2)
            start = time.time()

            s.sendto(b'Ping', (host, port))

            data, addr = s.recvfrom(1024)

            end = time.time()

            print(f"Received {data.decode()} from {addr} in {end - start:.2f} seconds")

        except socket.timeout:
            print("Request timed out")

if __name__ == "__main__":
    ping_server()
